 577| /*=========================================================================
 578|  * 十四、Timer0 中断：100us 时基 + 软件 PWM + 蜂鸣器方波 + 10ms 节拍
 579|  *   using 1 换一组寄存器，这样进中断就不用存主循环的 R0~R7，省点时间。
 580|  *   正常一趟中断就两次加法一次比较，P0 每 5 个 tick 才重新组装一次。
 581|  *   一趟中断必须比 99.83us 短，不然就丢 tick，PWM 直接乱套。
 582|  *=========================================================================*/
 583| void timer0_isr(void) interrupt 1 using 1
 584| {
 585|     unsigned char a;
 586| 
 587|     /* ---- ① 100us 系统时基：红外脉宽测量、各种超时都靠它，必须每次都加 ---- */
 588|     if (++tick_l == 0)
 589|         tick_h++;
 590| 
 591|     /* ---- ② PWM：每 PWM_DIV 个 tick 走一步 ---- */
 592|     if (++pwm_div >= PWM_DIV) {
 593|         pwm_div = 0;
 594| 
 595|         if (++pwm_cnt >= PWM_STEPS)
 596|             pwm_cnt = 0;
 597| 
 598|         ena = (pwm_cnt < duty_a);
 599|         enb = (pwm_cnt < duty_b);
 600| 
 601|         /* ---- ③ 蜂鸣器方波：每个 PWM 步（500us）翻一次 = 1kHz ---- */
 602|         if (buz_left) {
 603|             buz_left--;
 604|             buz_quiet = ~buz_quiet;
 605|         } else {
 606|             buz_quiet = 1;                  /* 半周期用完 -> 静音 */
 607|         }
 608| 
 609|         /* ---- ④ 组装 P0（主循环永不写 P0，只改上面这几个变量）---- */
 610|         a = 0x80;                           /* P0.7 恒 1（未使用，且 P0 是开漏口） */
 611|         a |= dirbits;                       /* D0~D3 = IN1~IN4 方向 */
 612|         if (ena) a |= 0x10;                 /* P0.4 = ENA */
 613|         if (enb) a |= 0x20;                 /* P0.5 = ENB */
 614|         if (buz_quiet) a |= 0x40;           /* P0.6 = 1 -> 静音 */
 615|         P0 = a;
 616|     }
 617|

 619|      * 放在写 P0 之后，对 PWM 时序没有影响。
 620|      * 必须放在中断里：主循环一趟最长会被收红外堵 25.6ms，用它数节拍会有
 621|      * 很大误差，录制回放的时长就不准了。 */

 623|         ms_div = 0;
 624|         if (++ms10_l == 0)

 630|      * 主循环只负责发触发和取结果，一次都不等。
 631|      * 开销很小：空闲时绝大多数 tick 只走一条判断就过去了。 */
 632|     if (us_st != US_ST_IDLE) {
 633|         if (us_st == US_ST_WAIT) {
 634|             if (US_ECHO) {                  /* 上升沿到了：开始数宽度 */
 635|                 us_wid = 0;
 636|                 us_st  = US_ST_HIGH;
 637|             } else if (us_wait < US_TOUT_TICKS) {
 638|                 us_wait++;                  /* 还没来，接着等（主循环不受影响）*/
 639|             } else {                        /* 超时：模块没响应这次触发 */

 641|                 us_res = US_FAR;
 642|                 us_got = 1;
 643| #if AV_SCOPE
 644|                 us_stage = 0;               /* 自测版 E：等不到上升沿 */
 645| #endif
 646|             }
 647|         } else {                            /* US_ST_HIGH：数回波宽度 */
 648|             if (US_ECHO) {
 649|                 if (us_wid < US_RANGE) {
 650|                     us_wid++;               /* 还没够远，接着数 */
 651|                 } else {                    /* 已经够远，不必再等下降沿 */
 652|                     us_st  = US_ST_IDLE;
